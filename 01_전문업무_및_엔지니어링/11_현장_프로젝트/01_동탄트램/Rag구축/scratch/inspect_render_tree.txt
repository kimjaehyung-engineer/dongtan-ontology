1851:   function classifyDoc(docName) {
1852:     let zone = '05. 기타 및 일반';
1853:     let field = '기타 · 일반';
1854: 
1855:     // Zone
1856:     if (docName.includes('1공구') || docName.startsWith('NH')) {
1857:       zone = '01. 1공구 본선';
1858:     } else if (docName.includes('2공구') || docName.startsWith('DT')) {
1859:       zone = '02. 2공구 본선';
1860:     } else if (docName.includes('차량기지') || docName.startsWith('GB') || docName.startsWith('NGB')) {
1861:       zone = '03. 차량기지';
1862:     } else if (docName.includes('제안') || docName.includes('증빙') || docName.includes('#3편')) {
1863:       zone = '04. 기술제안 및 특화';
1864:     } else if (docName.includes('입찰') || docName.includes('안내서') || docName.includes('과업') || docName.includes('공정표') || docName.includes('총괄')) {
1865:       zone = '00. 총괄 및 입찰·계약';
1866:     }
1867: 
1868:     // Field
1869:     if (docName.includes('토질') || docName.includes('지반') || docName.includes('시추') || docName.includes('기초') || docName.includes('비탈면') || docName.includes('가시설') || docName.includes('사면')) {
1870:       field = '토목 · 지반';
1871:     } else if (docName.includes('궤도') || docName.includes('분기기') || docName.includes('레일')) {
1872:       field = '궤도 · 철도';
1873:     } else if (docName.includes('건축') || docName.includes('정거장') || docName.includes('환승') || docName.includes('검수고') || docName.includes('관리동')) {
1874:       field = '건축 · 정거장';
1875:     } else if (docName.includes('신호') || docName.includes('전기') || docName.includes('통신') || docName.includes('전차선') || docName.includes('시스템')) {
1876:       field = '시스템 (신호/전기/통신)';
1877:     } else if (docName.includes('입찰') || docName.includes('계약') || docName.includes('총괄')) {
1878:       field = '총괄 · 계약';
1879:     }
1880: 
1881:     return { zone, field };
1882:   }
